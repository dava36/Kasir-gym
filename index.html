function playMoneySound() { try { const AudioContext = window.AudioContext || window.webkitAudioContext; if (!AudioContext) return; const ctx = new AudioContext(); // Resume AudioContext jika dalam keadaan suspended if (ctx.state === 'suspended') { ctx.resume(); } const tones = [987.77, 1318.51, 1758.40, 2637.02]; tones.forEach((freq, index) => { const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = 'sine'; osc.frequency.setValueAtTime(freq, ctx.currentTime + index * 0.08); gain.gain.setValueAtTime(0.2, ctx.currentTime + index * 0.08); gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + index * 0.08 + 0.3); osc.connect(gain); gain.connect(ctx.destination); osc.start(ctx.currentTime + index * 0.08); osc.stop(ctx.currentTime + index * 0.08 + 0.3); }); } catch (e) { console.log("Audio error:", e); } }
